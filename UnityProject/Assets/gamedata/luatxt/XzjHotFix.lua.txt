xlua.private_accessible(CS.XiuZhenJie.ZongMen_NeiMenWnd)
xlua.hotfix(CS.XiuZhenJie.ZongMen_NeiMenWnd,'NaJieGoodsChoosed',function(self,num,gameObj,data)   	    
	local item = gameObj:GetComponent("UI_GoodsItem")
	cast(item,typeof(CS.XiuZhenJie.UI_GoodsItem))
	local residual = data.GoodsNum - num
	if(residual<=0)
	then
	 num = data.GoodsNum
	 CS.UnityEngine.GameObject.Destroy(gameObj)
	end
	if(self.m_ChooseGoodsDic:ContainsKey(data.GoodsId) and self.m_ChooseGoodsItemDic:ContainsKey(data.GoodsId))
	then
	 local goodNum = 1
	 for k,v in pairs(self.m_ChooseGoodsDic)
	 do
	  if(k == data.GoodsId)
	  then
	  goodNum =v.GoodsNum + num
	  v.GoodsNum = goodNum
	  break
	  end
	 end
	 for k,v in pairs(self.m_ChooseGoodsItemDic)
	 do
	  if(k == data.GoodsId)
	  then
	  v:ResetNum(goodNum)
	  break
	  end
	 end
	else
	local baseNaJieGoodEntity = data:Copy()
	baseNaJieGoodEntity.GoodsNum = num
	local uI_GoodsItem = self:CreateObjItem(baseNaJieGoodEntity, self.Panel.JuanZengChooseContent,function(obj,gameObj,aa)
    self:ChooseContentGoodsClick(obj,gameObj,aa) 
	end)
	self.m_ChooseGoodsDic:Add(baseNaJieGoodEntity.GoodsId, baseNaJieGoodEntity)				
	self.m_ChooseGoodsItemDic:Add(baseNaJieGoodEntity.GoodsId, uI_GoodsItem)
	end
	self:RemoveNaJieDic(data.GoodsId, num)
    self:SetContentHight()
end)
